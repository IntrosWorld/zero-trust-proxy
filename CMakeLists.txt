cmake_minimum_required(VERSION 3.20)
project(zt_proxy VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ZT_ASAN "Address and undefined sanitizers" OFF)
option(ZT_TSAN "Thread sanitizer" OFF)
option(ZT_FUZZ "Build libFuzzer targets" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(ZT_ASAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
if(ZT_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=thread -no-pie)
endif()
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
file(GLOB_RECURSE CORE CONFIGURE_DEPENDS src/*.cpp)
list(FILTER CORE EXCLUDE REGEX "main.cpp$")
add_library(zt_core ${CORE})
target_include_directories(zt_core PUBLIC src)
target_link_libraries(zt_core PUBLIC Threads::Threads OpenSSL::Crypto)
add_executable(zt-proxy src/main.cpp)
target_link_libraries(zt-proxy PRIVATE zt_core)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest QUIET)
  if(NOT GTest_FOUND)
    include(FetchContent)
    FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.tar.gz)
    FetchContent_MakeAvailable(googletest)
  endif()
  file(GLOB UNIT CONFIGURE_DEPENDS tests/unit/*.cpp)
  if(UNIT)
    add_executable(unit_tests ${UNIT})
    target_link_libraries(unit_tests PRIVATE zt_core GTest::gtest_main)
    target_compile_definitions(unit_tests PRIVATE ZT_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
    add_test(NAME unit COMMAND unit_tests)
  endif()
endif()
install(TARGETS zt-proxy RUNTIME DESTINATION bin)
if(ZT_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Fuzzing requires Clang")
  endif()
  target_compile_options(zt_core PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
  target_link_options(zt_core INTERFACE -fsanitize=address,undefined)
  foreach(name http sni)
    add_executable(fuzz_${name} tests/fuzz/${name}_fuzz.cpp)
    target_compile_options(fuzz_${name} PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(fuzz_${name} PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_libraries(fuzz_${name} PRIVATE zt_core)
  endforeach()
endif()
option(ZT_BENCH "Build component benchmark" OFF)
if(ZT_BENCH)
  add_executable(body_bench bench/body_bench.cpp)
  target_link_libraries(body_bench PRIVATE zt_core)
endif()
